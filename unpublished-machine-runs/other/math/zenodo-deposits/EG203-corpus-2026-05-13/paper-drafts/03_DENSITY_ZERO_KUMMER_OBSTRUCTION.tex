\title{A Density-Zero Kummer Obstruction Theorem for Erd\H{o}s--Graham Problem \#203}
\author{Jared Wilder}
\date{Draft 0.1 --- 2026-05-12}

\begin{document}

\begin{abstract}
This draft formulates the safe Erd\H{o}s--Graham \#203-adjacent theorem extracted from the Kummer obstruction program.  It does not claim a full solution of Problem \#203.  Instead, it isolates the exact local subgroup obstruction
\[
\Gamma_d=\langle2,3\rangle\subset(\Z/d\Z)^*
\]
and records a density-zero theorem template: the centered local obstruction has exact second moments over \(m\), and after inserting subgroup-size lower bounds of Pappalardi type and a standard second-moment sieve, the possible \(m\) for which the Kummer local obstruction dominates form a density-zero family.  The final submission must make the sieve functional and exceptional set fully explicit; this draft supplies the theorem architecture and proof skeleton.
\end{abstract}

\section{Honest status}

This paper is not a proof that for every \((m,6)=1\),
\[
|\{(k,\ell):m2^k3^\ell+1\text{ prime}\}|=\infty.
\]
That full assertion remains conditional on a high-\(\ell\) Kummer non-concentration theorem.

The safe theorem is weaker: for \(100\%\) of \(m\), the exact Kummer local obstruction is variance-controlled.

\section{Local obstruction}

For squarefree \(d\), set
\[
G_d=(\Z/d\Z)^*,\qquad \Gamma_d=\langle2,3\rangle\subset G_d.
\]
For \(m\in G_d\), define
\[
\rho_m(d)=\frac{\1_{\Gamma_d}(-m^{-1})}{|\Gamma_d|}
\]
and
\[
\Delta_m(d)=\rho_m(d)-\frac1{\varphi(d)}.
\]

\begin{lemma}[Exact moments]\label{L:moments}
For squarefree \(d\) with \((d,6)=1\),
\[
\sum_{m\in G_d}\Delta_m(d)=0,
\]
and
\[
\sum_{m\in G_d}|\Delta_m(d)|^2=
\frac1{|\Gamma_d|}-\frac1{\varphi(d)}.
\]
\end{lemma}

\begin{proof}
The map \(m\mapsto -m^{-1}\) is a bijection of \(G_d\).  The result is the variance identity for the normalized indicator of the subgroup \(\Gamma_d\subset G_d\).
\end{proof}

\begin{definition}[Kummer obstruction energy]
Define
\[
\mathcal E(d)=\frac1{|\Gamma_d|}-\frac1{\varphi(d)}.
\]
Then Lemma \ref{L:moments} says
\[
\mathcal E(d)=\sum_{m\in G_d}|\Delta_m(d)|^2.
\]
\end{definition}

\section{Prime-level weighted variance}

For a finite set of primes \(\mathcal Q\), let \(P=\prod_{q\in\mathcal Q}q\).

\begin{lemma}[Weighted variance]\label{L:weighted}
For weights \(\lambda_q\),
\[
\E_{m\in(\Z/P\Z)^*}
\left|
\sum_{q\in\mathcal Q}\lambda_q\Delta_m(q)
\right|^2
=
\sum_{q\in\mathcal Q}
|\lambda_q|^2
\frac1{q-1}
\left(
\frac1{|\Gamma_q|}-\frac1{q-1}
\right).
\]
\end{lemma}

\begin{proof}
Expanding the square, diagonal terms are Lemma \ref{L:moments}.  Cross terms vanish by the Chinese remainder theorem and the mean-zero identity.
\end{proof}

\section{Subgroup-size input}

The required arithmetic input is a lower bound for
\[
|\Gamma_q|=|\langle2,3\rangle\bmod q|
\]
for all but a controlled set of primes.  A theorem of Pappalardi type gives lower bounds for the order of subgroups generated by multiplicatively independent rationals.

\begin{hypothesis}[Pappalardi-type order input]\label{H:papp}
There exist constants \(\eta>0\), \(C_1,C_2>0\), and an exceptional prime set \(\mathcal E_{\mathrm{ord}}(X)\) satisfying
\[
|\mathcal E_{\mathrm{ord}}(X)|\le C_1\frac{X}{(\log X)^{1+C_2}}
\]
or another summable density bound, such that for primes \(q\le X\), \(q\notin\mathcal E_{\mathrm{ord}}(X)\),
\[
|\langle2,3\rangle\bmod q|\ge q^\eta.
\]
\end{hypothesis}

\begin{remark}
The final paper must replace Hypothesis \ref{H:papp} by the exact quoted theorem, including constants and the precise exceptional-set statement.
\end{remark}

\section{Density-zero theorem template}

Let \(M\to\infty\), and let \(\mathcal Q=\mathcal Q(M)\) be a prime set with associated weights \(\lambda_q\).  Define the obstruction sum
\[
\mathcal D_m(\mathcal Q)=\sum_{q\in\mathcal Q}\lambda_q\Delta_m(q).
\]

\begin{proposition}[Variance control]\label{P:variance}
Assuming the order input above and suitable support conditions on \(\lambda_q\),
\[
\E_{m\le M,(m,6P)=1}|\mathcal D_m(\mathcal Q)|^2=o(1)
\]
after normalization by the relevant main term.
\end{proposition}

\begin{proof}[Proof skeleton]
Reduce the \(m\le M\) average to residue classes modulo \(P\), with an error controlled by the support size.  Apply Lemma \ref{L:weighted}.  Use
\[
\frac1{|\Gamma_q|}-\frac1{q-1}\le\frac1{|\Gamma_q|}
\]
and the Pappalardi-type lower bound outside the exceptional prime set.  Choose \(\mathcal Q\) and \(\lambda_q\) so that the resulting diagonal sum is \(o(1)\) after normalization.
\end{proof}

\begin{theorem}[Density-zero Kummer obstruction theorem; final-form placeholder]\label{T:dz}
Let \(E(M)\) be the set of \(m\le M\), \((m,6)=1\), for which the Kummer local obstruction sum exceeds the second-moment sieve threshold.  Under the explicit Pappalardi input and the standard second-moment sieve normalization specified above,
\[
|E(M)|=o(M).
\]
\end{theorem}

\begin{proof}[Proof skeleton]
Apply Proposition \ref{P:variance} and Chebyshev's inequality to the normalized obstruction sum.  The resulting exceptional set has size \(o(M)\).  The threshold and normalization must be specified in the final paper so that the theorem states exactly what the proof controls.
\end{proof}

\section{What this theorem does and does not say}

It does say:
\[
\text{the exact Kummer subgroup obstruction is density-zero exceptional in }m.
\]

It does not say:
\[
\text{Erd\H{o}s--Graham \#203 is solved for every }m.
\]

The full statement for every \(m\) requires a separate Kummer prime-distribution theorem.
