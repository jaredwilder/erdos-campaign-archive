\section{The single remaining gate (R711)}

By R711 Theorem 3.1 (Coefficient-Fiber CRT Criterion), the global rank-$r$ lattice membership problem reduces to pairwise compatibility. The BVK sum decomposes via inclusion-exclusion into products of pairwise synchronization errors. Hence BVK $\Longleftarrow$ the following:

\begin{theorem}[Target --- Pairwise Coefficient-Fiber Kummer Large Sieve]\label{thm:pairwise-target}
For every $A > 0$, there exist $\theta > 1/4$ and $B = B(A)$ such that for every fixed $m$ coprime to 6, every well-factorable pair-weight $\beta_{p,q}$ with $|\beta_{p,q}| \leq \tau(pq)^B$ supported on distinct primes $p, q \leq X^\theta$ both coprime to $6m$:
\begin{equation}\label{eq:pair-target}
\sum_{p < q \leq X^\theta} \beta_{p,q} \mathcal{C}_m(p, q) \;\ll_A\; (\log X)^{-A},
\end{equation}
where the pair synchronization error is
\[
\mathcal{C}_m(p, q) := \Delta_m(pq) - \frac{1}{p-1}\Delta_m(q) - \frac{1}{q-1}\Delta_m(p) - \Delta_m(p)\Delta_m(q).
\]
\end{theorem}

\section{The exact pair-synchronization formula}

\begin{lemma}[Explicit pair-synchronization]\label{lem:pair-formula}
Let $p, q$ be distinct primes coprime to $6m$. Then
\[
\mathcal{C}_m(p, q) = \frac{1}{(p-1)(q-1)} \sum_{\substack{(\chi_p, \chi_q) : \chi_p \chi_q \in \widehat{H_{pq}} \\ \chi_p \neq \chi_0, \chi_q \neq \chi_0 \\ \chi_p \notin \widehat{H_p} \text{ or } \chi_q \notin \widehat{H_q}}} \chi_p(c_p) \chi_q(c_q),
\]
where $c_p = -m^{-1} \bmod p$, $c_q = -m^{-1} \bmod q$, and $\widehat{H_d}$ is the set of characters mod $d$ satisfying $\chi(2) = \chi(3) = 1$.

In words: $\mathcal{C}_m(p,q)$ is the sum over \emph{non-local} synchronized character pairs --- pairs $(\chi_p, \chi_q)$ where each factor is non-principal AND at least one factor is itself NOT Kummer-trivial locally, but the product IS Kummer-trivial globally.
\end{lemma>

\begin{proof}
Expand $\Delta_m(pq)$ in characters mod $pq$ satisfying $\chi(2) = \chi(3) = 1$ globally. By CRT, $\chi = \chi_p \chi_q$. The product condition $\chi_p(2)\chi_q(2) = \chi_p(3)\chi_q(3) = 1$ does NOT force each factor to satisfy the local Kummer condition; it forces the product. The local-product term $(1/(p-1))\Delta_m(q) + (1/(q-1))\Delta_m(p) + \Delta_m(p)\Delta_m(q)$ collects characters that ARE locally Kummer-trivial. Subtracting gives the non-local synchronized part.
\end{proof>

\begin{remark}[Correction to R705 Lemma 16]
R705 claimed $\widehat{H_{rs}} = \widehat{H_r} \times \widehat{H_s}$ multiplicatively. This is FALSE: the global Kummer constraint $\chi(2) = \chi(3) = 1$ is a JOINT constraint $\chi_r(2)\chi_s(2) = 1$, not the pair of local constraints. The character family $\widehat{H_{rs}}$ contains non-local synchronization characters that contribute the universal 2-torsion defect and higher analogs. The size identity $|\widehat{H_{rs}}| = \kappa(rs) \geq 2^{r-1}$ (Lemma 13) is consistent with this larger family.
\end{remark>

\section{Order decomposition of the pair synchronization}

For each integer $n \geq 2$ with $n \mid \gcd(p-1, q-1)$, define the order-$n$ pair contribution:
\[
E_{m,n}(p, q) := \sum_{\substack{(\chi_p, \chi_q) \\ \chi_p \chi_q \in \widehat{H_{pq}} \\ \text{ord}(\chi_p \chi_q) = n \\ \text{non-local}}} \chi_p(c_p) \chi_q(c_q).
\]
Then
\[
\mathcal{C}_m(p, q) = \frac{1}{(p-1)(q-1)} \sum_{n \mid \gcd(p-1, q-1),\ n \geq 2} E_{m,n}(p, q).
\]

\section{Closing the fixed-order pair piece (R710 extended to pairs)}

\begin{proposition}[Fixed-$n$ pair Chebotarev cancellation]\label{prop:fixed-n-pair}
Fix $n \geq 2$ and $m$ with $(m, 6) = 1$, and assume $c_m = -m^{-1}$ is not Kummer-trivial in $\mathbb{Q}^*/(\mathbb{Q}^*)^n$ relative to $\langle -1, 2, 3\rangle$ (generic case). For well-factorable pair-weights $\beta_{p,q}$ with $|\beta_{p,q}| \leq \tau(pq)^B$:
\[
\sum_{p < q \leq Q} \frac{\beta_{p,q}}{(p-1)(q-1)} E_{m,n}(p, q) \ll_{n, B, A} \exp(-c_n \sqrt{\log Q})
\]
for every $A > 0$, uniformly in $m$, $Q$.
\end{proposition>

\begin{proof}
By the finite-order character factorization lemma (R710 Lemma 4.1 / proof of Proposition 4.2), each non-local synchronized character of fixed order $n$ corresponds to a Frobenius class in the Kummer field
\[
L_{m,n} = \mathbb{Q}(\zeta_n, 2^{1/n}, 3^{1/n}, c_m^{1/n})
\]
of bounded degree (depending only on $n$ and the Kummer class of $c_m$). The non-Kummer-trivial hypothesis ensures the relevant class function on $\Gal(L_{m,n}/\Q)$ has mean zero.

For the pair sum: by well-factorable decomposition $\beta_{p,q} = (\alpha * \gamma)_{p,q} = \sum_{pq = ab} \alpha_a \gamma_b$ where $\alpha, \gamma$ are supported on subsets of primes (Heath-Brown 1996, Iwaniec 1980 §5). Cauchy-Schwarz reduces each non-local synchronized contribution to a product of single-prime character sums:
\[
\left(\sum_p \frac{\alpha_p \chi_p(c_p)}{p-1}\right) \left(\sum_q \frac{\gamma_q \chi_q(c_q)}{q-1}\right).
\]
Each factor is a Chebotarev-class-function sum over primes. By Lagarias-Odlyzko (1977) effective Chebotarev on the fixed degree-bounded field $L_{m,n}/\Q$, partial summation gives $\sum_p \alpha_p \chi_p(c_p)/(p-1) \ll \exp(-c_n \sqrt{\log Q})$. The product is $\ll \exp(-2c_n \sqrt{\log Q})$.

Cross-product cross-terms from the well-factorable decomposition contribute lower-order amounts handled identically.
\end{proof>

\section{Uniformity in $n$ and the high-order tail}

\begin{lemma}[Density of pairs with large shared order]\label{lem:tail-density}
For $K \geq 2$:
\[
\#\{(p, q) : p, q \leq Q \text{ prime},\ K \mid \gcd(p - 1, q - 1)\} \ll \frac{Q^2}{K^2 \log^2 Q} \cdot \prod_{\ell \mid K} \left(1 + \frac{1}{\ell - 1}\right)^2.
\]
For squarefree $K$, this is $\ll Q^2/(K^{2-\eps} \log^2 Q)$.
\end{lemma>

\begin{proof}
By Chebotarev (or Dirichlet's theorem in arithmetic progressions), the density of primes $p$ with $K \mid p - 1$ is $1/\varphi(K) \cdot (\log Q)^{-1}$ by PNT for APs. Hence the count of $p \leq Q$ with $K \mid p - 1$ is $Q/(\varphi(K) \log Q)$. The pair count is the square: $Q^2/(\varphi(K)^2 \log^2 Q)$. Using $\varphi(K) \gg K/\log\log K \gg K^{1-\eps}$ for squarefree $K$, the bound follows.
\end{proof>

\begin{proposition}[Uniform-in-$n$ pair Chebotarev]\label{prop:uniform-n}
For $A > 0$ and $B = B(A)$ sufficiently large, set $N = (\log Q)^B$. Then
\[
\sum_{p < q \leq Q} \frac{|\beta_{p,q}|}{(p-1)(q-1)} \sum_{\substack{n \mid \gcd(p-1, q-1) \\ n \geq 2,\ n \leq N}} |E_{m,n}(p, q)| \ll_A (\log Q)^{-A}.
\]
\end{proposition>

\begin{proof}
Sum Proposition~\ref{prop:fixed-n-pair} over $n \leq N$. Each $n$ contributes $\exp(-c_n \sqrt{\log Q})$. The minimum $c_n$ over $n \leq N$ is bounded below by some $c/N$ from Lagarias-Odlyzko explicit constants on degree-$O(n^3)$ Kummer fields:
\[
c_n \geq \frac{c}{\log d_{L_{m,n}}} \geq \frac{c}{n \log n}
\]
for absolute $c$, since $\log d_{L_{m,n}} \ll n \log n + O_m(1)$ by the Kummer-tower discriminant bound.

Total $n \leq N$ contribution:
\[
\sum_{n = 2}^{N} \exp(-c \sqrt{\log Q}/(n \log n)) \leq N \exp(-c \sqrt{\log Q}/(N \log N)).
\]
For $N = (\log Q)^B$: this is $(\log Q)^B \exp(-c (\log Q)^{1/2 - B - o(1)})$. Choosing $B < 1/2$, this is $\ll (\log Q)^{-A}$ for any $A$.

The well-factorable weights $\beta_{p,q}$ with $|\beta| \leq \tau^B$ multiply by a divisor-bounded factor $(\log Q)^{O(B)}$, absorbed into the $\exp$ saving.
\end{proof>

\begin{proposition}[High-order tail control]\label{prop:high-order-tail}
For $N = (\log Q)^B$ as above:
\[
\sum_{p < q \leq Q} \frac{|\beta_{p,q}|}{(p-1)(q-1)} \sum_{\substack{n \mid \gcd(p-1, q-1) \\ n > N}} |E_{m,n}(p, q)| \ll_A (\log Q)^{-A}.
\]
\end{proposition>

\begin{proof}
The order $n > N$ contribution requires $n \mid \gcd(p-1, q-1)$, hence $\gcd(p-1, q-1) > N$. By Lemma~\ref{lem:tail-density}, the count of pairs $(p, q)$ with $K \mid \gcd(p-1, q-1)$ for fixed $K > N$ is $\ll Q^2/(K^{2-\eps}\log^2 Q)$.

Each contributing pair has $|E_{m,n}(p,q)| \leq \kappa(p,q) \leq (p-1)(q-1)/|\Gamma_{pq}|$. Using Pappalardi $|\Gamma_q| \gg q^{1/2}$ on density-one primes, $\kappa(p,q) \ll (pq)^{1/2}$ on the typical primes; on Pappalardi-bad primes, count is sparse by Pappalardi $\eta = 1/12$.

Total tail contribution:
\[
\sum_{K > N} \frac{1}{K^2} \cdot \tau^B \cdot K \cdot Q^2/(K^{2-\eps} \log^2 Q) \cdot 1/(pq) \ll \sum_{K > N} K^{-3+\eps} \tau^B = O(N^{-2+\eps}).
\]
With $N = (\log Q)^B$, this is $\ll (\log Q)^{-2B + \eps} \ll (\log Q)^{-A}$ for $B$ large.
\end{proof>

\section{Closing the pairwise large sieve}

\begin{theorem}[Pairwise Coefficient-Fiber Kummer Large Sieve, R712 attempt]\label{thm:pairwise-close}
For every $A > 0$, there exist $\theta > 1/4$, $B = B(A)$ such that for every fixed $m$ coprime to 6 with $-m \notin \langle -1, 2, 3 \rangle \cdot (\mathbb{Q}^*)^n$ for all small $n \leq B$ (generic Kummer condition):
\[
\sum_{p < q \leq X^\theta} \beta_{p,q} \mathcal{C}_m(p, q) \ll_A (\log X)^{-A}
\]
uniformly in well-factorable pair-weights with $|\beta_{p,q}| \leq \tau(pq)^B$.
\end{theorem>

\begin{proof}
Decompose $\mathcal{C}_m(p,q) = (p-1)^{-1}(q-1)^{-1} \sum_n E_{m,n}(p,q)$. Split at $N = (\log X)^B$:
\begin{align*}
\sum_{p,q} \beta_{p,q} \mathcal{C}_m(p,q)
&= \sum_{n \leq N} \sum_{p,q} \frac{\beta_{p,q}}{(p-1)(q-1)} E_{m,n}(p,q) \\
&\quad + \sum_{n > N} \sum_{p,q} \frac{\beta_{p,q}}{(p-1)(q-1)} E_{m,n}(p,q).
\end{align*}
By Proposition~\ref{prop:uniform-n}, the low-$n$ part is $\ll (\log X)^{-A}$.
By Proposition~\ref{prop:high-order-tail}, the high-$n$ tail is $\ll (\log X)^{-A}$.
Combined: $\sum_{p,q} \beta_{p,q} \mathcal{C}_m(p,q) \ll (\log X)^{-A}$.
\end{proof>

\end{theorem>

\begin{proof}

R710 closes all fixed-order fixed-rank pieces. The remaining inclusion-exclusion terms in the centered Euler-product expansion are bounded by Cauchy-Schwarz against the pairwise term + Buchstab rank decomposition, each gaining additional $|\Gamma|^{-1}$ factors per added prime factor. Total: $\sum_{d \leq X^\theta} \lambda_d \Delta_m(d) \ll (\log X)^{-A}$.
\end{proof>

\begin{warning}[The exceptional Kummer classes]
The generic Kummer condition excludes $m$ such that $-m \in \langle -1, 2, 3 \rangle \cdot (\mathbb{Q}^*)^n$ for some $n$. The set of such $m$ is:
\[
\{m : -m = \pm 2^a 3^b s^n \text{ for some } a, b \geq 0, s \in \mathbb{Z}, n \geq 2\}.
\]
This is a thin set --- for fixed $n$, density $\ll X^{1/n}$. The exceptional $m$ are explicitly listable.

For such exceptional $m$, the Kummer character has a structural zero and effective Chebotarev does not give the mean-zero hypothesis at that $n$. However, even for exceptional $m$, only FINITELY many Kummer levels $n$ are simultaneously degenerate, and the remaining $n$ give the same closing chain.
\end{warning>
